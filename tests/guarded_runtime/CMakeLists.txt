cmake_minimum_required(VERSION 3.31)
project(GuardedRuntimeWorkerTest LANGUAGES C CXX)
if(NOT PACK_SOURCE OR NOT EXISTS "${PACK_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "PACK_SOURCE must name the frozen complete candidate source")
endif()
set(MGC_WITH_NCNN OFF CACHE BOOL "" FORCE)
set(MGC_VULKAN OFF CACHE BOOL "" FORCE)
set(MGC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PACK_SOURCE}" pack)
target_compile_definitions(meguviz_compute_core PRIVATE MGC_TEST_GUARDED_HOST=1)
add_executable(guarded_runtime_worker runtime_worker_test.cpp)
target_compile_features(guarded_runtime_worker PRIVATE cxx_std_17)
option(MGC_RUNTIME_LEGACY_SNAPSHOT "Build comparison against pre-waiter-introspection runtime" OFF)
if(MGC_RUNTIME_LEGACY_SNAPSHOT)
  target_compile_definitions(guarded_runtime_worker PRIVATE MGC_RUNTIME_LEGACY_SNAPSHOT=1)
endif()
if(MSVC)
  set_property(TARGET guarded_runtime_worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
target_include_directories(guarded_runtime_worker PRIVATE "${PACK_SOURCE}/private")
target_link_libraries(guarded_runtime_worker PRIVATE MeguvizCompute::core)
enable_testing()
foreach(mode IN ITEMS success guard_first sibling failed unknown cancelled cancel_failed private_cancel closing race two_deps hol_skip waiter_reuse cancel_window publish_before publish_after ordinary_after publish_race ticket_after_close latency)
  add_test(NAME guarded_runtime_${mode} COMMAND guarded_runtime_worker ${mode})
  set_tests_properties(guarded_runtime_${mode} PROPERTIES TIMEOUT 30)
endforeach()
