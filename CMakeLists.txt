set(MGC_GLSLANG_COMMIT "780b6a209132dc74e99458f62ebe5fa7e57c41cf")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/locks/ncnn.lock.json" MGC_NCNN_LOCK)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/locks/ncnn.lock.json")
string(JSON MGC_PRIVATE_SDK_REVISION GET "${MGC_NCNN_LOCK}" private_sdk_revision)

if(MGC_ANDROID_STATIC_OPENMP AND (NOT ANDROID OR NOT MGC_WITH_NCNN))
  message(FATAL_ERROR "Android static OpenMP requires Android and the ncnn provider")

if(MGC_WITH_NCNN)
  if(NOT EXISTS "${MGC_NCNN_SOURCE}/src/net.h")
    message(FATAL_ERROR "MGC_NCNN_SOURCE must point to the complete pinned ncnn checkout")
  endif()
  execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}" -C "${MGC_NCNN_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_ncnn OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual_ncnn STREQUAL MGC_NCNN_COMMIT)
    message(FATAL_ERROR "ncnn source commit mismatch: ${actual_ncnn}")
  endif()
  string(JSON MGC_NCNN_PATCH_HASH GET "${MGC_NCNN_LOCK}" patches 0 sha256)
  execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}" -C "${MGC_NCNN_SOURCE}" diff --binary -- src cmake CMakeLists.txt OUTPUT_FILE "${CMAKE_CURRENT_BINARY_DIR}/ncnn.actual.patch" COMMAND_ERROR_IS_FATAL ANY)
  file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/ncnn.actual.patch" actual_patch_hash)
  if(NOT actual_patch_hash STREQUAL MGC_NCNN_PATCH_HASH)
    message(FATAL_ERROR "ncnn patch ledger mismatch: ${actual_patch_hash}")
  endif()
  set(NCNN_VERSION 20261020 CACHE STRING "Pinned source release version" FORCE)
  set(NCNN_VULKAN ${MGC_VULKAN} CACHE BOOL "" FORCE)
  set(NCNN_INSTALL_SDK ON CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(NCNN_OPENMP ${MGC_ANDROID_STATIC_OPENMP} CACHE BOOL "" FORCE)
  set(NCNN_BUILD_WITH_STATIC_CRT ON CACHE BOOL "" FORCE)
  if(MGC_ANDROID_STATIC_OPENMP)
    # ncnn's own lookup is optional.  An ON identity must never silently
    # describe a build whose compiler did not actually enable OpenMP.
    set(NCNN_SIMPLEOMP OFF CACHE BOOL "" FORCE)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    if(NOT OpenMP_CXX_FOUND OR NOT TARGET OpenMP::OpenMP_CXX)
      message(FATAL_ERROR "Android static OpenMP compiler support is unavailable")
    endif()
    if(NOT ANDROID_ABI MATCHES "^(arm64-v8a|armeabi-v7a)$" OR NOT CMAKE_CXX_COMPILER_TARGET)
      message(FATAL_ERROR "Locked Android OpenMP supports the two ARM ABIs with an explicit compiler target")
    endif()
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" "--target=${CMAKE_CXX_COMPILER_TARGET}" --print-file-name=libomp.a
      OUTPUT_VARIABLE MGC_OPENMP_RUNTIME OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT IS_ABSOLUTE "${MGC_OPENMP_RUNTIME}" OR NOT EXISTS "${MGC_OPENMP_RUNTIME}")
      message(FATAL_ERROR "Android NDK static OpenMP runtime not found")
    endif()
    get_filename_component(MGC_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(MGC_OPENMP_NOTICE "${MGC_COMPILER_BIN}/../NOTICE" ABSOLUTE)
    if(NOT EXISTS "${MGC_OPENMP_NOTICE}")
      message(FATAL_ERROR "Android NDK LLVM notice not found")
    endif()
    file(SHA256 "${MGC_OPENMP_RUNTIME}" MGC_OPENMP_RUNTIME_SHA256)
    file(SHA256 "${MGC_OPENMP_NOTICE}" MGC_OPENMP_NOTICE_SHA256)
    string(JSON MGC_EXPECTED_OPENMP_RUNTIME GET "${MGC_NCNN_LOCK}" android_openmp runtime_sha256 "${ANDROID_ABI}")
    string(JSON MGC_EXPECTED_OPENMP_NOTICE GET "${MGC_NCNN_LOCK}" android_openmp notice_sha256)
    if(NOT MGC_OPENMP_RUNTIME_SHA256 STREQUAL MGC_EXPECTED_OPENMP_RUNTIME OR
       NOT MGC_OPENMP_NOTICE_SHA256 STREQUAL MGC_EXPECTED_OPENMP_NOTICE)
      message(FATAL_ERROR "Android NDK OpenMP runtime or notice differs from the locked identity")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MGC_OPENMP_RUNTIME}" "${MGC_OPENMP_NOTICE}")
  endif()
  if(MGC_VULKAN)
    execute_process(COMMAND git -c "safe.directory=${MGC_GLSLANG_SOURCE}" -C "${MGC_GLSLANG_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_glslang OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual_glslang STREQUAL MGC_GLSLANG_COMMIT)
      message(FATAL_ERROR "glslang source commit mismatch")
    endif()
    set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
    set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
    set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
    set(GLSLANG_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
    if(NOT EXISTS "${MGC_NCNN_SOURCE}/glslang/CMakeLists.txt")
      message(FATAL_ERROR "Initialize the pinned glslang submodule before a Vulkan build")
    endif()
    execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}/glslang" -C "${MGC_NCNN_SOURCE}/glslang" rev-parse HEAD OUTPUT_VARIABLE nested_glslang OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT nested_glslang STREQUAL MGC_GLSLANG_COMMIT)
      message(FATAL_ERROR "ncnn glslang submodule identity mismatch")
    endif()
    set(NCNN_SYSTEM_GLSLANG OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${MGC_NCNN_SOURCE}" ncnn)
  if(MGC_ANDROID_STATIC_OPENMP)
    get_target_property(MGC_NCNN_COMPILE_OPTIONS ncnn COMPILE_OPTIONS)
    get_target_property(MGC_NCNN_LINK_OPTIONS ncnn INTERFACE_LINK_LIBRARIES)
    if(NOT "-fopenmp" IN_LIST MGC_NCNN_COMPILE_OPTIONS OR
       NOT "-fopenmp" IN_LIST MGC_NCNN_LINK_OPTIONS OR
       NOT "-static-openmp" IN_LIST MGC_NCNN_LINK_OPTIONS)
      message(FATAL_ERROR "ncnn did not enable the required static Android OpenMP compile and link options")
    endif()
  endif()
  # ncnn's SIMD options apply to its own translation units. Public C ABI entries
  # always catch C++ exceptions before returning across the boundary.
endif()
